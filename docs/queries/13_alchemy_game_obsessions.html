-- 13_alchemy_game_obsessions.sql
-- Purpose: Surface user prompts that pre-figure the current alchemy beat-em-up
-- discussion. Looks for the intersection of alchemy/lab/hermetic vocabulary
-- with game/mechanic/sprite/hazard vocabulary. Returns conversation titles,
-- date, and a snippet from the prompt.
-- Expected runtime: 5-15s.
-- Expected rows: ~150-400.

WITH hits AS (
  SELECT
    conversation_id,
    conversation_title,
    source_name,
    DATE(created_at) AS day,
    approx_words,
    SUBSTR(prompt_text, 1, 400) AS snippet,
    -- score: weighted hit count
    (CASE WHEN LOWER(prompt_text) LIKE '%alchem%'   THEN 1 ELSE 0 END +
     CASE WHEN LOWER(prompt_text) LIKE '%hermet%'   THEN 1 ELSE 0 END +
     CASE WHEN LOWER(prompt_text) LIKE '%vitriol%'  THEN 1 ELSE 0 END +
     CASE WHEN LOWER(prompt_text) LIKE '%athanor%'  THEN 1 ELSE 0 END +
     CASE WHEN LOWER(prompt_text) LIKE '%alembic%'  THEN 1 ELSE 0 END +
     CASE WHEN LOWER(prompt_text) LIKE '%emblem%'   THEN 1 ELSE 0 END) AS alch_score,
    (CASE WHEN LOWER(prompt_text) LIKE '%beat-em-up%' OR LOWER(prompt_text) LIKE '%beat em up%' OR LOWER(prompt_text) LIKE '%beatem%' THEN 3 ELSE 0 END +
     CASE WHEN LOWER(prompt_text) LIKE '%streets of rage%' OR LOWER(prompt_text) LIKE '%king of the monsters%' THEN 3 ELSE 0 END +
     CASE WHEN LOWER(prompt_text) LIKE '%fighting game%' THEN 2 ELSE 0 END +
     CASE WHEN LOWER(prompt_text) LIKE '%roguelike%'  THEN 2 ELSE 0 END +
     CASE WHEN LOWER(prompt_text) LIKE '%board game%' THEN 2 ELSE 0 END +
     CASE WHEN LOWER(prompt_text) LIKE '%lab accident%' OR LOWER(prompt_text) LIKE '%cascading%' THEN 2 ELSE 0 END +
     CASE WHEN LOWER(prompt_text) LIKE '%hazard%'    THEN 1 ELSE 0 END +
     CASE WHEN LOWER(prompt_text) LIKE '%sprite%'    THEN 1 ELSE 0 END +
     CASE WHEN LOWER(prompt_text) LIKE '%pixel art%' THEN 1 ELSE 0 END +
     CASE WHEN LOWER(prompt_text) LIKE '%game mechanic%' OR LOWER(prompt_text) LIKE '%mechanic%' THEN 1 ELSE 0 END) AS game_score
  FROM pa_prompts
  WHERE
    (LOWER(prompt_text) LIKE '%alchem%'
       OR LOWER(prompt_text) LIKE '%hermet%'
       OR LOWER(prompt_text) LIKE '%vitriol%'
       OR LOWER(prompt_text) LIKE '%athanor%'
       OR LOWER(prompt_text) LIKE '%alembic%'
       OR LOWER(prompt_text) LIKE '%emblem%')
    AND
    (LOWER(prompt_text) LIKE '%game%'
       OR LOWER(prompt_text) LIKE '%mechanic%'
       OR LOWER(prompt_text) LIKE '%sprite%'
       OR LOWER(prompt_text) LIKE '%hazard%'
       OR LOWER(prompt_text) LIKE '%accident%'
       OR LOWER(prompt_text) LIKE '%roguelike%'
       OR LOWER(prompt_text) LIKE '%board game%'
       OR LOWER(prompt_text) LIKE '%beat%')
    AND char_count > 80
    AND char_count < 3000
)
SELECT
  day,
  source_name,
  conversation_title,
  alch_score,
  game_score,
  (alch_score + game_score) AS total_score,
  approx_words,
  snippet
FROM hits
WHERE (alch_score + game_score) >= 3
ORDER BY total_score DESC, day ASC
LIMIT 200;
